megoldas: |
  nevek = ["Kovács Anna", "Nagy Péter", "Szabó Dóra", "Tóth Gábor", "Varga Lilla"]
  f1 = [12, 8, 15, 10, 14]
  f2 = [15, 14, 15, 6, 9]
  f3 = [9, 14, 13, 11, 15]

  print("1. feladat")
  osszes = 0
  for i in range(len(nevek)):
      osszes += f1[i] + f2[i] + f3[i]
  print("Versenyzők: " + str(len(nevek)) + ", összes pont: " + str(osszes))

  print("2. feladat")
  vegeredmeny = []
  for i in range(len(nevek)):
      vegeredmeny.append(f1[i] + f2[i] + f3[i])
      print(nevek[i] + ": " + str(vegeredmeny[i]) + " pont")

  print("3. feladat")
  monogramok = []
  for nev in nevek:
      m = ""
      for szo in nev.split(" "):
          m += szo[0] + ". "
      monogramok.append(m.strip())
  print("Monogramok: " + ", ".join(monogramok))

  print("4. feladat")
  legjobb = max(vegeredmeny)
  elol = []
  for i in range(len(nevek)):
      if vegeredmeny[i] == legjobb:
          elol.append(nevek[i])
  if len(elol) == 1:
      print("Győztes: " + elol[0] + " (" + str(legjobb) + " pont)")
  else:
      print("Holtverseny az élen: " + ", ".join(elol) + " (" + str(legjobb) + " pont)")

  print("5. feladat")
  atlagok = [sum(f1) / len(f1), sum(f2) / len(f2), sum(f3) / len(f3)]
  jo = 0
  for k in range(3):
      if atlagok[k] > atlagok[jo]:
          jo = k
  print("A legjobb átlagú forduló: " + str(jo + 1) + ". (" + str(round(atlagok[jo], 1)) + ")")

  print("6. feladat")
  kuszob = int(input())
  van = False
  for i in range(len(nevek)):
      if f1[i] >= kuszob and f2[i] >= kuszob and f3[i] >= kuszob:
          print(nevek[i])
          van = True
  if not van:
      print("Senki.")
